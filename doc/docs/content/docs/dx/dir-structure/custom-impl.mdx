This is a guide for the internals of the `dir-structure` library.
After reading this, you should be able to understand a bit about how the
library works, and how to extend it. In this guide, we will be extending
it to support a new type of wrapper, `FsMtime`, which will allow us to
read and write the modification time of files.

 ## !!steps Write the mtime

```rust ! code.rs no_run
// !lints
use std::path::Path;
use std::time::SystemTime;

use dir_structure::{ReadFrom, Result, WrapIoError, WriteTo};

// !collapse(1:4) collapsed
struct FsMtime<T> {
    mtime: SystemTime,
    value: T,
}

// !collapse(1:9) collapsed
impl<T: ReadFrom> ReadFrom for FsMtime<T> {
    fn read_from(path: &Path) -> Result<Self> {
        let mtime = path.metadata()
            .and_then(|m| m.modified())
            .wrap_io_error_with(path)?;
        let value = T::read_from(path)?;
        Ok(FsMtime { mtime, value })
    }
}

// !collapse(1:9)
impl<T: WriteTo> WriteTo for FsMtime<T> {
    fn write_to(&self, path: &Path) -> Result<()> {
        self.value.write_to(path)?;
        // !mark(1:3)
        std::fs::File::open(path)
            .and_then(|f| f.set_modified(self.mtime))
            .wrap_io_error_with(path)?;
        Ok(())
    }
}
```

 ## !!steps End

And that's it! We have implemented a new wrapper type that allows us to read
and write the modification time of files. Let's see how to use this wrapper
in practice.

```rust ! code.rs no_run
// !lints
use std::path::Path;
use std::time::SystemTime;

use dir_structure::{DirStructure, DirStructureItem, ReadFrom, Result, WrapIoError, WriteTo};

// !collapse(1:4)
struct FsMtime<T> {
    mtime: SystemTime,
    value: T,
}

// !collapse(1:9)
impl<T: ReadFrom> ReadFrom for FsMtime<T> {
    fn read_from(path: &Path) -> Result<Self> {
        let mtime = path.metadata()
            .and_then(|m| m.modified())
            .wrap_io_error_with(path)?;
        let value = T::read_from(path)?;
        Ok(FsMtime { mtime, value })
    }
}

// !collapse(1:9)
impl<T: WriteTo> WriteTo for FsMtime<T> {
    fn write_to(&self, path: &Path) -> Result<()> {
        self.value.write_to(path)?;
        std::fs::File::open(path)
            .and_then(|f| f.set_modified(self.mtime))
            .wrap_io_error_with(path)?;
        Ok(())
    }
}

// !mark(1:9)
// !collapse(1:5)
#[derive(DirStructure)]
struct Dir {
    #[dir_structure(path = "file.txt")]
    file: FsMtime<String>,
}

// !hidden let path = "dir";
let mut dir = Dir::read(path)?;
dir.file.mtime = SystemTime::UNIX_EPOCH;
dir.write(path)?;
// !tail dir_structure::Error

```
