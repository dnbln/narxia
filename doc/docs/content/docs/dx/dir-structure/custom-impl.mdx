This is a guide for the internals of the `dir-structure` library.
After reading this, you should be able to understand a bit about how the
library works, and how to extend it. In this guide, we will be extending
it to support a new type of wrapper, `FsMtime`, which will allow us to
read and write the modification time of files.

 ## !!steps End

And that's it! We have implemented a new wrapper type that allows us to read
and write the modification time of files. Let's see how to use this wrapper
in practice.

```rust ! code.rs no_run
// !lints
// !req-feature derive
use std::path::Path;
use std::pin::Pin;
use std::time::SystemTime;

use dir_structure::{DirStructure, traits::sync::DirStructureItem, error::VfsResult, traits::vfs, error::WrapIoError, prelude::*};

// !collapse(1:4)
struct FsMtime<T> {
    mtime: SystemTime,
    value: T,
}

// !collapse(1:9)
impl<'vfs, Vfs: vfs::Vfs<'vfs, Path=Path>, T: ReadFrom<'vfs, Vfs>> ReadFrom<'vfs, Vfs> for FsMtime<T> {
    fn read_from(path: &Path, vfs: Pin<&'vfs Vfs>) -> VfsResult<Self, Vfs> {
        let mtime = path.metadata()
            .and_then(|m| m.modified())
            .wrap_io_error_with(path)?;
        let value = T::read_from(path, vfs)?;
        Ok(FsMtime { mtime, value })
    }
}

// !collapse(1:9)
impl<'vfs, Vfs: vfs::WriteSupportingVfs<'vfs, Path=Path>, T: WriteTo<'vfs, Vfs>> WriteTo<'vfs, Vfs> for FsMtime<T> {
    fn write_to(&self, path: &Path, vfs: Pin<&'vfs Vfs>) -> VfsResult<(), Vfs> {
        self.value.write_to(path, vfs)?;
        std::fs::File::open(path)
            .and_then(|f| f.set_modified(self.mtime))
            .wrap_io_error_with(path)?;
        Ok(())
    }
}

// !mark(1:9)
// !collapse(1:5)
#[derive(DirStructure)]
struct Dir<Vfs: vfs::VfsCore<Path=Path>> {
    #[dir_structure(path = "file.txt")]
    file: FsMtime<String>,
    __marker: std::marker::PhantomData<Vfs>,
}

// !hidden let path = "dir";
let mut dir = Dir::read(path)?;
dir.file.mtime = SystemTime::UNIX_EPOCH;
dir.write(path)?;
// !tail dir_structure::error::Error::<std::path::PathBuf>

```
